# ========== WEB FRAMEWORK ==========
# Framework web moderno y de alto rendimiento para construir APIs REST
fastapi
# Servidor ASGI recomendado para FastAPI (con autoreload y WebSockets)
uvicorn[standard]

# ========== LLM & AI FRAMEWORK ==========
# Framework principal de LangChain para agentes y workflows
langchain
# Componentes internos compartidos de LangChain
langchain-core
# Separadores de texto optimizados para procesamiento de documentos
langchain-text-splitters
# Integración oficial con OpenAI
langchain-openai
# Integración con Google Gemini
langchain-google-genai
# Integraciones mantenidas por la comunidad
langchain-community

# ========== LLM PROVIDERS ==========
# SDK oficial de OpenAI
openai
# Modelos de embeddings locales (MiniLM, MPNet, BGE, etc.)
sentence-transformers

# ========== ORCHESTRATION & WORKFLOWS ==========
# Framework de orquestación de agentes y workflows
langgraph
# Framework para supervisión y coordinación de múltiples agentes
langgraph-supervisor
# Persistencia de estado y checkpoints para LangGraph
langgraph-checkpoint
# Persistencia SQLite específica para LangGraph
langgraph-checkpoint-sqlite
# Observabilidad, tracing y debugging para LangGraph
langsmith

# ========== DATABASE & PERSISTENCE ==========
# ORM moderno para Python (SQLAlchemy 2.0)
sqlalchemy

chromadb
# Integración oficial LangChain ↔ Chroma
langchain-chroma

# ========== USER INTERFACE ==========
# Framework web para aplicaciones de datos en Python
streamlit
# Componentes de chat para Streamlit (opcional)
streamlit-chat

# ========== HTTP CLIENTS & NETWORKING ==========
# Cliente HTTP síncrono (requests)
requests

pydantic
# Validación de correos electrónicos
email-validator

pypdf

beautifulsoup4

# ========== UTILITIES & HELPERS ==========
# Carga de variables de entorno desde archivos .env
python-dotenv
# Reintentos y backoff exponencial para llamadas API
tenacity

# ========== THREAT INTELLIGENCE & SECURITY TOOLS ==========

vt-py
# Búsqueda web inteligente para agentes AI
tavily-python

# ========== TESTING ==========
# Framework de testing moderno para Python
pytest
